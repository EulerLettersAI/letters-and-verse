# خمسون عاماً بين أسوار الكلامْ

### (شهود)

خمسون عاماً بين أسوار الكلامْ

ما حان بعدُ الصومُ عن ذاك الفضول!

فمتى يحين!

أنْ تحظى بالشرفِ الرفيعْ

فلتنتظرْ كي تفهمَ العِشقَ الأصيلْ

أو مُتْ و لمْ تفهمْ سوى ذاك الفضولْ

حبّاتُ رملِ العِشقِ في كفّيك كانتْ دائماً...

حباتِ رمْلٍ ليسَ غيرْ

فمٓتى تكون حقيقةً و متى تكونْ..

ذَهباً مصفىً من شِراك المشركينْ!

خمسونَ عاماً بين أسوارِ الكلامْ

اصْمتْ فقدْ أخفقتَ في فهمٍ المُرادْ

<!--more-->

وتركتَ كلّ الأشقياءْ

و ركبتَ موجَ الغاضبينَ على نصوصِ الأولياءْ

مع أن قبطانَ السفينةِ لمْ يكنْ...

يوماً سوى عبدِ الأنا

لمْ يحظَ بالشرفِ الرفيعْ...

شرفِ الفقيرِ على بلاطِ الأولياءْ

### (دينوسياسة)

خمسون عاماً بين أسوارِ الكلامْ

لن أكتب الشعر المُقفى ثانيهْ

يكفي الذي خلَّفتُ يوماً من حُطامْ..

يحتاج للكنسِ الطويلْ..

من فَرطِ تصفيقِ البغايا

لكنني أعمى كثيراً ثُمّ أُبصرُ ثانيهْ

هل تكفي تلك الثانيهْ؟

مع كل ذيّاك الخرابْ!

هل كفَّ تصفيقُ البغايا؟

مِن أجلِ عرضِ الآي في الأسواقْ..

لِتُسامَ بالأحلامِ و الأنغامْ

حتى تترجمها المآقي و الدموعْ

بين الحقيقةِ و الهُراءْ

بينَ البكاء على المعاني و المبانيْ

وتهكّمِ الشَهَواتِ في أرضِ الخَفاءِ من الصدورْ

في جُعبةِ العصرِ الجديدْ

عصرِِ المباني و القصورْ

### (تجديد)

خمسون عاماً بين أسوار الكلامْ

هل يقنعُ الدرويش بالصمتِ المريحْ؟

أم أنّهُ آلى صعودَ المقصلهْ

ليبوءَ بالتقريع و التشنيعْ

و سبابِ مَن يهوى الكلامْ

والحُكمِ بالإعدامْ

هل كانَ أحرى أنْ يصانعَ في العجاجْ؟

و يصيحَ في الضوضاءِ شأنَ الصائحينْ

ويحيدَ عنْ أسماءِ كلِّ الأشقياءْ

لِيُعدَّ بينَ الصالحينْ

أمْ كانَ مأموراً بنصبِ المقصلهْ

ليُغلّبَ الخوفَ المريعَ على الرجاءْ

مِن أجلِ تحرير العقولِ من الثناء

وعبادةِ الأحبارِ و الرهبانْ

ليفيقَ بعضُ السادرين..

مِن سطوةِ التاريخْ

لمّا ترآى الشيخُ فوقَ المقصلهْ

### (خيانة)

خمسون عاماً بين أسوار الكلام

وبناتُ أفكارِ المريدِ كما يُريدْ

لايرعوي عنْ فحشهِ المدفونْ..

في قلبهِ المأخوذ بالأشكالْ

مرآته البلهاءُ تمنحهُ الكلامْ

ليصيحَ ملء الصوتِ أنتمْ كافرونْ

وأنا و أصحابي فقطْ...

على دربِ الفلاحْ

فاركبْ إذن معهمْ لتنجوَ مِن صياحْ

أوْ لُذْ بجيشٍ آخرٍ من صائحينْ

وتراشقوا الصيحاتْ

وتمرّغوا بحلاوةِ الإنكارْ

وبصحبةِ النُظّارْ

ولتذْبحوا الدرويش فوق المقصلهْ

فكلامهُ مِنْ قلبهِ و كلامكمْ...

لمْ يعدُ عنْ طرفِ اللسانْ

### (ضلال)

خمسون عاما بين أسوار الكلام

والقردُ يخطبُ كلّ يومٍ في العفافْ

 عيناه لا تَرَيانِ شيئاً مِنْ قفاهْ

 يحتاجُ مرآتين أوْ أكثرْ

 حتى يرى فِعلَ الخطيئةِ في قفاهْ

 و لأنّهُ قردُ القرودْ

 سيقولُ مرآتي خديعهْ

 لا تنطلي إلا على العميانْ

 مِنْ غيرِ فِرقتِنا السعيدةِ بالبصرْ

 فأنا أقيسُ لكمْ كما قاسَ الرجالْ

 فلتتركوا المرآةَ للنسوانْ

 ولتسمعوا خُطبَ الفضيلةِ كلّ يومْ

 ولتقذفوا كلَّ المرايا للجحيمْ

 و لترجموا الدنيا بأحجارِ العفافْ

 ثم استديروا باسمينْ

### (جدال)

خمسون عاماً بين أسوار الكلامْ

والصبيةُ الأغمارُ في أوجِ الغرورْ

حازوا على لَقَبِ البطولةِ في المِراءْ

لمْ يُثنِهم أنّ الفضائلَ تُحتَضرْ...

أوْ أنْ ينادي الشيخُ حيّ على الفلاحْ

فرعونُ واحدهمْ مِنَ البِلَّورْ..

أنّى يُرى إلا مِنَ الأطرافْ!

فليلبسوا حُللاً مِنَ الخذلانْ

وليحملوا ذهباً من العِصيانْ

وليكتبوا هذراً من الأشعارْ

إنّ الحقائق ثابتهْ...

لا تنقلبْ!

سوآتهمْ مكشوفةٌ في كلّ آنْ

إلا لأنفسهم ففي الأستارْ

هل يبصرون غداً على صفعِ الحياهْ

أم يبصرونَ غداً على قرعِ النعالْ

بعدَ المماتْ!

بعدَ الفواتْ!

### (انتفاخ)

خمسون عاما بين أسوار الكلامْ

و رجالنا صورٌ بلا معنى

لا يصبرونَ عن التبرّمِ كلّ يوم ..

يشكونَ من ضغطِ الحياهْ

و المدرسهْ!

والهاتفِ النقّالْ!

و الراتبِ الشهريِّ و التلفازْ!

يشكونَ مِن غنَجِ النساءْ

في هذه الأيّامْ

لا يصمتونَ عن الكلامِ بِلا فعالْ

و يُثرثرونَ على فضاءِ الحاسباتْ..

عَنْ أمّهاتٍ طاهياتٍ خادماتْ

خرّجنَ أشباهَ الرّجال!

والفاسدين!

كانت نساءُ الأمسِ تطحنُ بالرحى..

خبزاً لأفواهِ الرجالِ الصائحهْ

و تدقُ أعوادَ الشجرْ

وتلملمُ الزيتونْ

و تُهدهدُ الطفلَ الرضيعْ

لينامَ (سي السيّدْ) قريرَ العينْ

وتبشُّ في وجه الزعيمِ إذا أساءْ

حتى يشب الطفلُ في كَنفِ الأنا

تخشى على قارورةٍ مِنْ شاربَينْ

لكنّها قارورةٌ لا تستحي!

لتبادلِ الأدوارْ!

قارورةٌ لا تستفيقُ مِن الميوعة و الفسادْ

في شرقنا المنفوخِ بالأقوالْ

### (افتقار)

خمسون عاما بين اسوار الكلام

ها قدْ حُملتَ على السريرْ

عيناكَ تبتسمانِ في وجهِ السماءْ

هلْ تنظرانِ سوى هُنيههْ..

لتغادرا التكليفَ و الأغلالْ

لتغادرا هذا المكانْ..

هذا الزمانْ!

لتغادرا الوهْمَ المبطّنَ بالأنا..

أنّا رَمَينا باليمينْ..

دونَ المددْ!

أنّا لمسنا بالعلومْ

سرَّ الوجودْ!

عيناكَ تنتظرانِ شمساً لا تغيبْ

لتُرددا..

عَنتِ الوجوهْ

عَنتِ الوجوهْ

عَنتِ الوجوهْ

ليزولَ ذاك الإلتباسْ

فيْ ظُلمةِ العاداتْ..

بينَ الإرادةِ و الإرادهْ

بينَ الشقاوةِ و السعادهْ

و تعثّرِ الأشباحِ في معنى الوجودْ

حتى تُغني الروحْ..

عيناكَ لاشْ

فلْتصمتا

### (كبرياء)

خمسون عاما بين اسوار الكلام

و الطفلُ يكبرُ كلّ يومٍ راغماً..

لِيسفَّ مِنْ كَبِدِ الشّقاءْ..

قَلَقاً لِما يُبدي الصباحْ..

ولِما يُخبّيه المساءْ

يستلُّ بسْمةَ حالمٍ ما أمكنَهْ

مِنْ هذهِ الدنيا

لكنّه لا يرتوي..

لا يكتفي..

حتى يؤدّبهُ الفطامْ

أوْ يستحيلَ كما السباعِ شراهةً

ليطالَ مِن صَخَبِ الحياهْ..

ما أمكنه!

مِنْ كأسِ خمرٍ أو ذَهَبْ

أوْ بيتِ شِعرٍ أو مقالهْ

أوْ عِمّةٍ خضراءَ أوْ بيضاءَ أوْ سوداءْ

مِنْ أجلِ تصفيقِ الرجالْ

حتى يشيب الطفلُ يوماً..

في شرقِنا المجنونِ بالتصفيقْ

وبِحضْرَة (الدكتورْ)

وفضيلةِ الشيخِ الجليلْ

و سعادةِ النّوابْ

ووجوهِ أصحاب العشائرْ

وبكلِّ أسبابِ التفاخرْ

و لوِ اقتضى بعضَ الكَذِبْ

أوْ بعضَ تزييفِ الحقائقْ

للشاربينَ مِنَ البحارِ الزاخرهْ..

بالكبرياءْ!

### (مدد)

خمسون عاما بين أسوار الكلامْ

كمْ مرةٍ قلّبتَ طرْفكَ في السماءْ!

وذهلتَ عمّنْ يفتحُ الأبوابْ

يمشي كما يمشي البشرْ

مِنْ بينِ أظْهرِهِمْ ظَهَرْ

مِنْ نَسلِ آدمَ مِثْلنا

ما أقربَهْ!

بيديهِ مفتاحُ الفلاحْ

و ببابهِ تقفُ الرّجالْ..

مدَّ البصرْ

هذا يُجرجرُ زلّةً..

قدْ جاءه مستغفراً ربَّ السماءْ

هذا يلوذُ ببابهِ..

مِنْ فرطِ حُمّى أو شقاءْ

هذا يُداوي جُرحَهُ..

بتبادُلِ النَظراتْ

أو لَمسةٍ مِنْ كفهِ

فيذوبُ في يَدِهِ العَنَتْ

إنْ ذوَّبَ العَنَتُ الرِّجالْ

و تِهيْ تبثُّ ظُلامةً

كي تنجلي..

مِنْ نظرتهْ

ما أرأفهْ!

ما أرحمهْ!

يا قبلة الحيرانْ

يا راحة التعبانْ

يا ملجأ المضطرْ

يا أمنَنا في كلّ عصرْ

أرجو المددْ
